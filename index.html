Supabase Database Config

Enter your Supabase Project URL and Public Anon Key below. They are saved in your browser's local storage and used to fetch & sync your live database tables.

View SQL Setup Script for Supabase 
-- 1. Create Tables
CREATE TABLE IF NOT EXISTS public."CategoryType" (
  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  type text NOT NULL
);

CREATE TABLE IF NOT EXISTS public."Categories" (
  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  type bigint NOT NULL REFERENCES public."CategoryType"(id),
  subtype varchar NOT NULL
);

CREATE TABLE IF NOT EXISTS public."Transactions" (
  id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  date date NOT NULL,
  store varchar NOT NULL,
  description varchar,
  amount numeric NOT NULL,
  category bigint NOT NULL REFERENCES public."Categories"(id)
);

-- 2. Grants & RLS Policies
GRANT SELECT, INSERT, UPDATE, DELETE ON public."Transactions" TO anon, authenticated;
GRANT SELECT, INSERT, UPDATE, DELETE ON public."Categories" TO anon, authenticated;
GRANT SELECT, INSERT, UPDATE, DELETE ON public."CategoryType" TO anon, authenticated;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO anon, authenticated;

ALTER TABLE public."Transactions" ENABLE ROW LEVEL SECURITY;
ALTER TABLE public."Categories" ENABLE ROW LEVEL SECURITY;
ALTER TABLE public."CategoryType" ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow Transactions" ON public."Transactions" FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);
CREATE POLICY "Allow Categories" ON public."Categories" FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);
CREATE POLICY "Allow CategoryType" ON public."CategoryType" FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);
